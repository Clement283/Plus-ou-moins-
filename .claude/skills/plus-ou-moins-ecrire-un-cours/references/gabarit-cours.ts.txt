const ID = 'cNN-nom';
const RULE = 'À ÉCRIRE : la règle du cours en une phrase.';

// ——— Générateur (ADAPTER) : la réponse doit toujours être un entier juste ; niveau = taille des nombres ET longueur du calcul si le cours s'y prête ———
type Range = [number, number];
const RANGE: Record<Level, Range> = { 1: [2, 9], 2: [10, 30], 3: [10, 99] };
const ex1 = (a: number, b: number): Exercise => makeExercise(ID, 1, op('*', num(a), num(b)));
function generate(level: Level, rng: Rng): Exercise {
  const [na, nb] = signs(rng);
  const a = draw(rng, RANGE[level]);
  const b = draw(rng, RANGE[level]);
  return makeExercise(ID, level, op('*', num(na ? -a : a), num(nb ? -b : b)));
}
const parts2 = (ex: Exercise): [number, number] => operands(ex.expr) as [number, number];
function resolution(ex: Exercise): string[] {
  const [a, b] = parts2(ex).map(Math.abs) as [number, number];
  return [toLatex(ex.expr), `= ${sg(ex.answer.n)}(${a} \\times ${b})`, resultLine(ex)];
}

// ——— Erreurs reconnues (ADAPTER) : une ErrorSpec par erreur de la fiche ; `text` = message précis (jamais la règle générale) ———
const shown = (ex: Exercise): string => toText(ex.expr);
// partIndex = indice dans `parts` de l'écran qui enseigne la règle (ici 2 : accroche 0, idée 1, règle 1 = 2) ; label = le titre exact de cet écran
const RULE_1 = { n: 1, label: 'À ÉCRIRE : titre de la règle 1', partIndex: 2 };
const RULES: Record<string, NonNullable<ErrorDef['rule']>> = { 'regle-signes': RULE_1 };
const errors: ErrorDef[] = buildErrors(
  [
    {
      id: 'regle-signes',
      family: 'signe',
      detect: (ex, given) => ex.answer.n !== 0 && isOpposite(ex, given), // la réponse donnée est l'opposée de la bonne
      text: (ex) => `À ÉCRIRE : message précis pour ${shown(ex)} = ${txt(ex.answer.n)}.`,
      draw: (level, rng) => redraw(() => generate(level, rng), (ex) => ex.answer.n !== 0), // calcul-piège de cette erreur
      sample: (ex) => -ex.answer.n, // la réponse fausse typique
      wrongLines: (ex, w) => (w === -ex.answer.n ? stepped(ex, `${sg(w)}(…)`, w) : null), // résolution fausse affichée (ADAPTER)
    },
  ],
  resolution,
).map((e): ErrorDef => ({ ...e, rule: RULES[e.id] as NonNullable<ErrorDef['rule']> }));

// ——— Retours d'erreur des questions qui comptent : un message par erreur prévisible (voir catalogue : feedbackText) ———
function feedbackText(q: Question, o: Outcome): string | null {
  if (o.unfinished || o.errorFamily === 'ecriture') return null;
  return q.kind === 'saisie' && !o.correct ? `À ÉCRIRE : méthode pas à pas pour ${toText(q.ex.expr)}.` : null;
}
